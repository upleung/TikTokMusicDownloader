### 2. 数据库与去重模块 `database.py`

负责将抓取到的音乐入库，确保断点续传且不重复下载。

```python
import sqlite3
import os

class Database:
    def __init__(self, db_path="data/music.db"):
        os.makedirs(os.path.dirname(db_path), exist_ok=True)
        self.conn = sqlite3.connect(db_path)
        self.create_tables()

    def create_tables(self):
        cursor = self.conn.cursor()
        cursor.execute('''
            CREATE TABLE IF NOT EXISTS music (
                music_id TEXT PRIMARY KEY,
                title TEXT,
                artist TEXT,
                cover_url TEXT,
                play_url TEXT,
                video_url TEXT,
                downloaded INTEGER DEFAULT 0
            )
        ''')
        self.conn.commit()

    def insert_music(self, music_data):
        cursor = self.conn.cursor()
        cursor.execute('''
            INSERT OR IGNORE INTO music (music_id, title, artist, cover_url, play_url, video_url)
            VALUES (?, ?, ?, ?, ?, ?)
        ''', (music_data['music_id'], music_data['title'], music_data['artist'],
              music_data.get('cover_url', ''), music_data.get('play_url', ''), music_data.get('video_url', '')))
        self.conn.commit()

    def get_undownloaded(self):
        cursor = self.conn.cursor()
        cursor.execute('SELECT * FROM music WHERE downloaded = 0')
        return cursor.fetchall()

    def mark_downloaded(self, music_id):
        cursor = self.conn.cursor()
        cursor.execute('UPDATE music SET downloaded = 1 WHERE music_id = ?', (music_id,))
        self.conn.commit()

    def get_all(self):
        cursor = self.conn.cursor()
        cursor.execute('SELECT * FROM music')
        return cursor.fetchall()

```

### 3. 工具类模块 `utils.py`

处理文件名格式化及写入 ID3 音乐标签（封面、标题、作者）。

```python
import re
import requests
from mutagen.mp4 import MP4, MP4Cover

def sanitize_filename(filename):
    """去除文件名中不允许的特殊字符"""
    return re.sub(r'[\\/*?:"<>|]', "", filename).strip()

def write_id3_tags_m4a(file_path, title, artist, cover_url=None):
    """为 M4A 音频文件写入元数据标签"""
    try:
        audio = MP4(file_path)
        audio["\xa9nam"] = title  # 标题
        audio["\xa9ART"] = artist # 艺术家
        
        if cover_url:
            try:
                resp = requests.get(cover_url, timeout=10)
                if resp.status_code == 200:
                    # 识别图片格式
                    img_format = MP4Cover.FORMAT_PNG if 'png' in cover_url.lower() else MP4Cover.FORMAT_JPEG
                    audio["covr"] = [MP4Cover(resp.content, imageformat=img_format)]
            except Exception as e:
                print(f"  [!] 获取封面失败: {e}")
                
        audio.save()
    except Exception as e:
        print(f"  [!] 写入ID3标签失败: {e}")

```

### 4. 数据抓取模块 `collector.py`

**核心亮点**：不需要用户手动找 Cookie，启动后会打开浏览器，利用 Playwright 的网络拦截器静默抓取数据包。

```python
import json
import time
import os
import csv
from playwright.sync_api import sync_playwright
from database import Database

def extract_music_from_json(data):
    """递归提取 JSON 中的音乐信息（应对 TikTok 接口结构变化）"""
    found_music = []
    if isinstance(data, dict):
        if "id" in data and "title" in data and "playUrl" in data:
            found_music.append(data)
        else:
            for key, value in data.items():
                found_music.extend(extract_music_from_json(value))
    elif isinstance(data, list):
        for item in data:
            found_music.extend(extract_music_from_json(item))
    return found_music

def run_collector():
    os.makedirs("data", exist_ok=True)
    db = Database()
    music_list = []
    seen_ids = set()

    print("[*] 正在启动浏览器 (首次运行会下载内核，请稍候)...")
    with sync_playwright() as p:
        # 使用持久化上下文，保存登录状态
        user_data_dir = os.path.join(os.getcwd(), 'chrome_data')
        browser = p.chromium.launch_persistent_context(
            user_data_dir, 
            headless=False,
            viewport={"width": 1280, "height": 720}
        )
        page = browser.pages[0]

        def handle_response(response):
            url = response.url.lower()
            # 拦截包含 music 或 favorite 的 API 请求
            if "api" in url and ("favorite" in url or "music" in url or "item_list" in url):
                if response.request.method == "OPTIONS": 
                    return
                try:
                    data = response.json()
                    musics = extract_music_from_json(data)
                    for m in musics:
                        m_id = str(m.get("id", ""))
                        if m_id and m_id not in seen_ids:
                            title = m.get("title", "Unknown")
                            artist = m.get("authorName", m.get("ownerHandle", "Unknown"))
                            cover = m.get("coverLarge", m.get("coverMedium", m.get("coverThumb", "")))
                            play_url = m.get("playUrl", "")
                            
                            m_data = {
                                "music_id": m_id,
                                "title": title,
                                "artist": artist,
                                "cover_url": cover,
                                "play_url": play_url,
                                "video_url": ""
                            }
                            seen_ids.add(m_id)
                            music_list.append(m_data)
                            db.insert_music(m_data)
                            print(f"  [+] 捕获音乐: {title} - {artist}")
                except Exception:
                    pass
        
        page.on("response", handle_response)
        
        print("\n" + "="*60)
        print("请在弹出的浏览器中完成以下操作：")
        print("1. 登录你的 TikTok 账号 (若已登录请忽略)")
        print("2. 进入你的 个人主页 -> 点击【收藏 (Favorites)】 -> 点击【音乐 (Music)】")
        print("3. 当你看到你的 423 首音乐列表加载出来后")
        print("4. 回到此黑框终端，按下【回车键】开始自动向下滚动抓取！")
        print("="*60 + "\n")
        
        page.goto("https://www.tiktok.com/")
        input(">>> 准备好后，请按回车键开始自动滚动... ")

        print("[*] 开始自动向下滚动，请不要关闭浏览器 (预计需要1-3分钟)...")
        previous_height = 0
        retries = 0
        
        while True:
            # 自动滚到底部加载新数据
            page.evaluate("window.scrollTo(0, document.body.scrollHeight)")
            time.sleep(2.5) # 等待网络请求
            new_height = page.evaluate("document.body.scrollHeight")
            
            if new_height == previous_height:
                retries += 1
                if retries >= 4:
                    print("[*] 页面到底，未发现新内容，停止滚动。")
                    break
            else:
                retries = 0
            previous_height = new_height

        print(f"[*] 抓取完毕！本次运行共捕获到 {len(music_list)} 首音乐。")
        
        if music_list:
            # 保存至 JSON
            with open("data/music.json", "w", encoding="utf-8") as f:
                json.dump(music_list, f, ensure_ascii=False, indent=2)
            # 保存至 CSV
            with open("data/music.csv", "w", encoding="utf-8", newline='') as f:
                writer = csv.DictWriter(f, fieldnames=["music_id", "title", "artist", "cover_url", "play_url", "video_url"])
                writer.writeheader()
                writer.writerows(music_list)

        browser.close()

```

### 5. 音频下载模块 `downloader.py`

使用 `requests` 直接下载 或 `yt-dlp` 后备支持，保证高成功率。

```python
import os
import requests
import yt_dlp
from tqdm import tqdm
from database import Database
from utils import sanitize_filename, write_id3_tags_m4a

def run_downloader():
    os.makedirs("Music", exist_ok=True)
    db = Database()
    pending_music = db.get_undownloaded()
    all_music = db.get_all()

    if not pending_music:
        print("[*] 当前没有需要下载的音乐（已全部下载完毕）。")
        return

    print(f"[*] 发现 {len(pending_music)} 首待下载音乐，开始下载...")
    all_ids = [m[0] for m in all_music]

    for row in pending_music:
        music_id, title, artist, cover_url, play_url, video_url, _ = row
        
        try:
            index = all_ids.index(music_id) + 1
        except ValueError:
            index = 1

        safe_title = sanitize_filename(title)
        filename = f"{index:04d} - {safe_title}.m4a"
        filepath = os.path.join("Music", filename)
        
        print(f"\n[-] 正在处理: {filename}")
        success = False
        
        # 方式 1: 直接下载抓取到的直链
        if play_url and play_url.startswith("http"):
            try:
                headers = {
                    "User-Agent": "Mozilla/5.0 (Windows NT 10.0; Win64; x64) AppleWebKit/537.36",
                    "Referer": "https://www.tiktok.com/"
                }
                resp = requests.get(play_url, headers=headers, stream=True, timeout=15)
                if resp.status_code == 200:
                    total_size = int(resp.headers.get('content-length', 0))
                    with open(filepath, 'wb') as file, tqdm(
                        desc="下载进度", total=total_size, unit='iB', unit_scale=True, unit_divisor=1024
                    ) as bar:
                        for data in resp.iter_content(chunk_size=1024):
                            size = file.write(data)
                            bar.update(size)
                    success = True
            except Exception as e:
                print(f"  [!] 直链下载失败，尝试备用方案...")

        # 方式 2: 如果直链失效，利用 yt-dlp 根据 music_id 抓取 TikTok 音乐主页
        if not success:
            try:
                tiktok_music_url = f"https://www.tiktok.com/music/-{music_id}"
                filepath_no_ext = filepath.rsplit('.', 1)[0]
                ydl_opts = {
                    'format': 'bestaudio/best',
                    'outtmpl': filepath_no_ext + '.%(ext)s',
                    'quiet': True,
                    'postprocessors': [{
                        'key': 'FFmpegExtractAudio',
                        'preferredcodec': 'm4a',
                    }],
                }
                with yt_dlp.YoutubeDL(ydl_opts) as ydl:
                    ydl.download([tiktok_music_url])
                
                # yt-dlp 如果没有 ffmpeg 可能会保留原格式，确保重命名匹配
                if not os.path.exists(filepath):
                    for ext in ['.mp3', '.webm', '.mp4']:
                        if os.path.exists(filepath_no_ext + ext):
                            os.rename(filepath_no_ext + ext, filepath)
                            break
                success = True
            except Exception as e:
                print(f"  [!] yt-dlp 解析下载失败: {e}")

        if success and os.path.exists(filepath):
            write_id3_tags_m4a(filepath, title, artist, cover_url)
            db.mark_downloaded(music_id)
            print(f"  [+] 下载成功并已标记！")
        else:
            print(f"  [!] 该首音乐下载失败，将在下次运行时重试。")

```

### 💡 如何在本地运行与测试：

1. 将上面 7 个文件全部放在一个文件夹里（例如 `TikTokMusicDownloader`）。
2. **需要配置 FFmpeg（可选但强烈建议）**：为了让 `yt-dlp` 完美转换音频格式，建议去下载一个免安装版的 `ffmpeg.exe`，并把它放在和 `main.py` 同一个文件夹下。
3. 双击 `start.bat` 运行。
4. 弹出的浏览器属于“独立环境”，你需要**登录一次 TikTok 账号**（程序会自动保存登录状态 `chrome_data`，下次运行就不用登了）。
5. 按照黑框提示，点到你的“收藏音乐”界面，按下回车键，双手离开键盘，程序会自动帮你滚到底部提取所有 423 首歌的数据，随后全自动建立下载列队并写入精美的专辑封面和作者标签。
